% Mathematical Programming restricts the reference list to published/accepted
% work. These identifiers mention unpublished work directly in the text.
% Optional argument gives a locator in the explicitly identified version.
% The starred form omits the authors, for a citation right after their names.
\makeatletter
\newcommand{\preprintcite}{%
  \@ifstar{\bfgs@preprintcite\@gobble}{\bfgs@preprintcite\@firstofone}}
\newcommand{\bfgs@preprintcite}[1]{%
  \@ifnextchar[{\bfgs@@preprintcite#1}{\bfgs@@preprintcite#1[]}}
\def\bfgs@@preprintcite#1[#2]#3{%
  \@ifundefined{bfgspreprint@#3}%
    {\PackageError{bfgs-paper}{Unknown preprint key #3}{Define the verified source in preprint_references.tex.}}%
    {(#1{\csname bfgspreprintauthors@#3\endcsname, }%
      \csname bfgspreprint@#3\endcsname\if\relax\detokenize{#2}\relax\else, #2\fi)}%
}
