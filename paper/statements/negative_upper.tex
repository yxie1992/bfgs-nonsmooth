\begin{theorem}[Local failure with globally smooth convex branches]
\label{art:local-failure}
There exist four globally convex functions $f_i\in C^{2,1}(\R^2)$ satisfying
\[
 0\preceq\nabla^2f_i(x)\preceq64I\qquad(i=0,1,2,3;\;x\in\R^2)
\]
such that $F=\max_{0\le i\le3}f_i$ has compact sublevel sets and a unique
nonsmooth minimizer at $0$, together with starting points $x_0^{(\ell)}\to0$
($\ell\ge1$), all initialized with $H_0=I$, having the following property: for
every fixed pair
\[
 0<c_1\le\tfrac13,\qquad c_1<c_2<1,
\]
the unit-first bracketing BFGS algorithm of Section~\ref{art:failure} generates
the same infinite full-BFGS run from $x_0^{(\ell)}$, and that run satisfies
\[
 x_k^{(\ell)}\longrightarrow\bar x_\ell\ne0,\qquad
 F(x_k^{(\ell)})\downarrow F(\bar x_\ell)>F(0)=0,\qquad
 0\notin\partial F(\bar x_\ell).
\]
Every accepted and rejected trial point is differentiable, each line search
terminates in at most five trials with accepted step lengths repeating
$1,1/4,1/16$, and the supremum of the norms of all accepted and rejected query
points in the $\ell$th run tends to zero as $\ell\to\infty$.
\end{theorem}
