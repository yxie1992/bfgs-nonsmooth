\begin{theorem}[Local failure under uniform strong convexity]
\label{art:local-failure-lower}
There exist a finite strongly convex function $F=\max_{0\le i\le3}f_i$ on
$\R^2$ with compact sublevel sets and a unique nonsmooth minimizer at $0$,
where each branch $f_i\in C^{1,1}(\R^2)$ is uniformly strongly convex with
globally Lipschitz gradient and constants $0<\underline\mu\le\overline L<\infty$
satisfy
\[
 \underline\mu I\preceq\nabla^2F(x)\preceq\overline L I
 \quad\text{at every twice-differentiable point }x\in\R^2
\]
(with the first three branches not twice differentiable at $0$), together with
starting points $x_0^{(\ell)}\to0$ ($\ell\ge1$), all initialized with $H_0=I$,
such that for each fixed $\ell\ge1$, the infinite full-BFGS run from
$x_0^{(\ell)}$ generated by the unit-first bracketing algorithm of
Section~\ref{art:failure} is independent of the choice of $(c_1,c_2)$ for
$0<c_1\le1/3$, $c_1<c_2<1$ and satisfies
\[
 x_k^{(\ell)}\longrightarrow\bar x_\ell\ne0,\qquad
 \bar x_\ell\longrightarrow0,\qquad
 F(x_k^{(\ell)})\downarrow F(\bar x_\ell)>F(0),\qquad
 0\notin\partial F(\bar x_\ell).
\]
Every accepted and rejected trial point is differentiable, the accepted step
lengths repeat $1,1/4,1/16$, and the supremum of all query norms in the
$\ell$th run tends to zero as $\ell\to\infty$.
\end{theorem}
